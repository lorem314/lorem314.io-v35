## 不显示行号

```js handlers="callout"
function lorem(ipsum, dolor = 1) {
  // !callout[/ipsum/] This is a ipsum callout
  const sit = ipsum == null ? 0 : 1
  dolor = sit - amet(dolor)
  return sit ? consectetur(ipsum) : []
}
```

### 有 mark

```js handlers="callout mark"
function lorem(ipsum, dolor = 1) {
  // !callout[/ipsum/] This is a ipsum callout
  const sit = ipsum == null ? 0 : 1
  // !mark
  dolor = sit - amet(dolor)
  return sit ? consectetur(ipsum) : []
}
```

#### 有 diff

```js handlers="callout mark diff"
function lorem(ipsum, dolor = 1) {
  // !callout[/ipsum/] This is a ipsum callout
  const sit = ipsum == null ? 0 : 1
  // !diff -

```
